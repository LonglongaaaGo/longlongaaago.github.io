\cvsection{Selected Research \& Engineering Projects}
\role{UltraDiffEdit}{TNNLS 2026; first author}{Tuning-free high-resolution generative image editing}
\begin{points}
\item Developed multi-patch encoding, global-local consistency denoising, and progressive sampling to extend pretrained latent diffusion models to real-image editing without fine-tuning.
\item Demonstrated editing up to 8K resolution on a single RTX 3090. \href{https://github.com/LonglongaaaGo/UltraDiffEdit}{Code} and \href{https://longlongaaago.github.io/videos/}{video} are available.
\end{points}
\role{FACEMUG}{TVCG 2024; first author}{Multimodal local facial editing}
\begin{points}
\item Developed a unified editing framework for text, sketches, semantic maps, color maps, exemplars, and attribute labels, with multimodal feature fusion and self-supervised latent warping.
\item Preserves the unedited image regions while supporting local, controllable changes. Presented the work through the Genesis entrepreneur training program. \href{https://doi.org/10.1109/TVCG.2024.3434386}{Publication}.
\end{points}
\role{AdaMSS}{NeurIPS 2025; co-author}{Parameter-efficient fine-tuning}
\begin{points}
\item Co-authored adaptive multi-subspace updates for expressive and efficient adaptation of pretrained language and vision models.
\item The method is integrated into \href{https://github.com/huggingface/peft/tree/main/examples/adamss_finetuning}{Hugging Face PEFT}. \href{https://github.com/jzheng20/AdaMSS}{Research code} is public.
\end{points}
\role{DocPure}{TCSVT 2026; co-first author}{Prompt-free unified document restoration}
\begin{points}
\item Co-authored structure-guided wavelet modulation for restoration across document degradations without manual inference-time task prompts. \href{https://github.com/LingmingSSS/DocPure}{Code}.
\end{points}
\role{GRIG}{CVM 2025; first author}{Data-efficient generative residual image inpainting}
\begin{points}
\item Developed iterative residual inpainting using CNN features and Transformer reasoning for small training sets.
\item Introduced forgery-patch adversarial training to improve textures and fine detail. \href{https://doi.org/10.26599/CVM.2025.9450408}{Paper}; \href{https://github.com/LonglongaaaGo/GRIG_few_shot_inpainting}{code}.
\end{points}
\role{Visual Style Prompt Learning}{Pattern Recognition 2025; first author}{Diffusion-guided blind face restoration}
\begin{points}
\item Developed diffusion-generated visual style prompts to guide blind face restoration.
\item Designed style-modulated aggregation to fuse visual prompts with restoration features. \href{https://doi.org/10.1016/j.patcog.2024.111312}{Paper}; \href{https://github.com/LonglongaaaGo/VSPBFR}{code}.
\end{points}
\role{Local Agentic Code Editor}{Nasdaq internal prototype; 2026}{Exploratory side project: API-driven code editing and local tool orchestration}
\begin{points}
\item Prototyped a local AI code editor with an \textbf{agentic loop} that selects between code writing, code review, and shell-tool execution based on the current task.
\item Integrated \textbf{language-model APIs with local tool execution} to support code generation, debugging, and iterative review.
\end{points}
